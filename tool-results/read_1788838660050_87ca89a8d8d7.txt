    43→/** compuerta genérica de N entradas con opcional inversión de entradas */
    44→function gateEval(op: "AND" | "OR" | "XOR", negateOutput: boolean) {
    45→  return (ctx: EvalContext) => {
    46→    const neg = (ctx.props.negInputs as boolean[] | undefined) ?? [];
    47→    const vals = Object.keys(ctx.inputs)
    48→      .sort()
    49→      .map((k, i) => {
    50→        const v = n(ctx.inputs[k]);
    51→        return neg[i] ? (v ? 0 : 1) : v;
    52→      });
    53→    let out: number;
    54→    if (op === "AND") out = vals.every((v) => v === 1) ? 1 : 0;
    55→    else if (op === "OR") out = vals.some((v) => v === 1) ? 1 : 0;
    56→    else out = vals.reduce((a, b) => a ^ b, 0);
    57→    if (negateOutput) out = out ? 0 : 1;
    58→    ctx.outputs["out"] = out;
    59→  };
    60→}
    61→
    62→const gatePins = (props: Record<string, unknown>): PinDef[] => {
    63→  const count = (props.nIn as number) ?? 2;
    64→  return inPins(count).concat([{ id: "out", name: "out", direction: "output", width: 1 }]);
    65→};

    67→const negField = {
    68→  kind: "toggle" as const,
    69→  key: "negInputs",
    70→  label: "Negar entradas (burbujas)",

    73→const DELAYS: Record<string, number> = {
    74→  AND: 2, OR: 3, NOT: 1, NAND: 2, NOR: 2, XOR: 4, XNOR: 4, BUFFER: 1,
    75→};

    77→const delayField = (dflt: number) => ({
    78→  kind: "number" as const,
    79→  key: "delay",
    80→  label: "Retardo",
    81→  min: 0,
    82→  max: 50,
    83→  step: 0.5,
    84→  suffix: "ns",
    85→  ...{ default: dflt },
    86→});

    88→function gateTruth(name: string, nIn: number, op: (v: number[]) => number): TruthTable {
    89→  const rows: string[][] = [];
    90→  const total = 1 << nIn;
    91→  for (let m = 0; m < total; m++) {
    92→    const vals = Array.from({ length: nIn }, (_, i) => (m >> (nIn - 1 - i)) & 1);
    93→    rows.push([...vals.map(String), String(op(vals))]);
    94→  }
    95→  return { headers: [...Array.from({ length: nIn }, (_, i) => String.fromCharCode(65 + i)), "S"], rows };
    96→}
    97→
    98→const andOp = (v: number[]) => (v.every((x) => x === 1) ? 1 : 0);
    99→const orOp = (v: number[]) => (v.some((x) => x === 1) ? 1 : 0);
   100→const xorOp = (v: number[]) => v.reduce((a, b) => a ^ b, 0);
   101→
   102→function multiGateDef(
   103→  type: ComponentType,
   104→  label: string,
   105→  short: string,
   106→  op: "AND" | "OR" | "XOR",
   107→  negOut: boolean,
   108→  desc: string,
   109→): ComponentDef {
   110→  return {
   111→    type,
   112→    label,
   113→    short,
   114→    category: "gates",
   115→    description: desc,
   116→    defaults: { nIn: 2, delay: DELAYS[type as string] ?? 2, negInputs: [] },
   117→    propFields: [
   118→      { kind: "select", key: "nIn", label: "Entradas", options: [2, 3, 4, 5, 6, 7, 8].map((v) => ({ value: v, label: `${v}` })) },
   119→      delayField(DELAYS[type as string] ?? 2),
   120→      negField,
   121→    ],
   122→    pins: gatePins,
   123→    eval: gateEval(op, negOut),
   124→    truthTable: (p) => gateTruth(label, (p.nIn as number) ?? 2, op === "AND" ? andOp : op === "OR" ? orOp : xorOp),
   125→    equation: () => null,
   126→  };
   127→}
